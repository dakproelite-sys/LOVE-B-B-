<script type="module">
    import { auth, database } from "./love-bebe-service.js";
    import { onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-auth.js";
    import { ref, get, push } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-database.js";

    let currentUser = null;
    let userProfile = null;
    let cloudinaryConfig = null;
    let currentStream = null;
    let mediaRecorder = null;
    let recordedChunks = [];
    let capturedBlob = null;
    let isVideoMode = false;
    let facingMode = "user";
    let timerInterval = null;

    const cameraVideo = document.getElementById("cameraVideo");
    const capturedImagePreview = document.getElementById("capturedImagePreview");
    const capturedVideoPreview = document.getElementById("capturedVideoPreview");
    const timerBadge = document.getElementById("timerBadge");
    const timerText = document.getElementById("timerText");
    const takePhotoBtn = document.getElementById("takePhotoBtn");
    const recVideoBtn = document.getElementById("recVideoBtn");
    const switchCamBtn = document.getElementById("switchCamBtn");
    const publishBtn = document.getElementById("publishBtn");
    const statusMsg = document.getElementById("statusMsg");

    onAuthStateChanged(auth, async (user) => {
        if (!user) {
            window.location.replace("connexion.html");
            return;
        }
        currentUser = user;

        try {
            const configSnap = await get(ref(database, "config/cloudinary"));
            if (configSnap.exists()) cloudinaryConfig = configSnap.val();

            const userSnap = await get(ref(database, `users/${user.uid}`));
            if (userSnap.exists()) userProfile = userSnap.val();

            initCamera();
        } catch (e) {
            console.error(e);
        }
    });

    async function initCamera() {
        if (currentStream) {
            currentStream.getTracks().forEach(track => track.stop());
        }
        try {
            currentStream = await navigator.mediaDevices.getUserMedia({
                video: { facingMode: facingMode },
                audio: true
            });
            cameraVideo.srcObject = currentStream;
        } catch (err) {
            statusMsg.textContent = "Impossible d'accéder à la caméra.";
        }
    }

    switchCamBtn.onclick = () => {
        facingMode = facingMode === "user" ? "environment" : "user";
        initCamera();
    };

    // Prise de photo instantanée
    takePhotoBtn.onclick = () => {
        const canvas = document.createElement("canvas");
        canvas.width = cameraVideo.videoWidth || 640;
        canvas.height = cameraVideo.videoHeight || 480;
        const ctx = canvas.getContext("2d");
        ctx.drawImage(cameraVideo, 0, 0, canvas.width, canvas.height);

        canvas.toBlob((blob) => {
            capturedBlob = blob;
            isVideoMode = false;
            capturedImagePreview.src = URL.createObjectURL(blob);
            capturedImagePreview.style.display = "block";
            cameraVideo.style.display = "none";
            capturedVideoPreview.style.display = "none";
            publishBtn.disabled = false;
            statusMsg.textContent = "Photo capturée !";
        }, "image/jpeg", 0.9);
    };

    // Enregistrement vidéo 15 secondes
    recVideoBtn.onclick = () => {
        recordedChunks = [];
        try {
            mediaRecorder = new MediaRecorder(currentStream, { mimeType: "video/webm" });
        } catch (e) {
            mediaRecorder = new MediaRecorder(currentStream);
        }

        mediaRecorder.ondataavailable = (e) => {
            if (e.data.size > 0) recordedChunks.push(e.data);
        };

        mediaRecorder.onstop = () => {
            capturedBlob = new Blob(recordedChunks, { type: "video/webm" });
            isVideoMode = true;
            capturedVideoPreview.src = URL.createObjectURL(capturedBlob);
            capturedVideoPreview.style.display = "block";
            cameraVideo.style.display = "none";
            capturedImagePreview.style.display = "none";
            timerBadge.style.display = "none";
            publishBtn.disabled = false;
            statusMsg.textContent = "Vidéo de 15s enregistrée !";
        };

        mediaRecorder.start();
        timerBadge.style.display = "flex";
        let timeLeft = 15;
        timerText.textContent = `00:${timeLeft < 10 ? '0' : ''}${timeLeft}`;

        timerInterval = setInterval(() => {
            timeLeft--;
            timerText.textContent = `00:${timeLeft < 10 ? '0' : ''}${timeLeft}`;
            if (timeLeft <= 0) {
                clearInterval(timerInterval);
                if (mediaRecorder.state === "recording") mediaRecorder.stop();
            }
        }, 1000);
    };

    // Publication vers Cloudinary et Firebase
    publishBtn.onclick = async () => {
        if (!capturedBlob || !cloudinaryConfig) return;

        publishBtn.disabled = true;
        statusMsg.textContent = "Envoi du média vers Cloudinary...";

        const formData = new FormData();
        formData.append("file", capturedBlob, isVideoMode ? "video.webm" : "photo.jpg");
        formData.append("upload_preset", cloudinaryConfig.upload_preset);

        const resourceType = isVideoMode ? "video" : "image";

        try {
            const res = await fetch(`https://api.cloudinary.com/v1_1/${cloudinaryConfig.cloud_name}/${resourceType}/upload`, {
                method: "POST",
                body: formData
            });
            const data = await res.json();

            if (data.secure_url) {
                const newPost = {
                    userId: currentUser.uid,
                    userName: userProfile ? userProfile.fullName : "Membre",
                    userAvatar: userProfile && userProfile.profileMediaUrl ? userProfile.profileMediaUrl : "https://via.placeholder.com/42",
                    description: document.getElementById("postCaption").value.trim(),
                    theme: document.getElementById("postTheme").value,
                    mediaUrl: data.secure_url,
                    isVideo: isVideoMode,
                    createdAt: Date.now()
                };

                await push(ref(database, "gallery_posts"), newPost);
                statusMsg.textContent = "Publication réussie ! Redirection vers la galerie...";
                setTimeout(() => {
                    window.location.href = "photos.html";
                }, 1500);
            } else {
                statusMsg.textContent = "Échec du téléversement Cloudinary.";
                publishBtn.disabled = false;
            }
        } catch (err) {
            console.error(err);
            statusMsg.textContent = "Erreur de réseau lors de la publication.";
            publishBtn.disabled = false;
        }
    };
</script>
